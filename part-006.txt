function BankPage() {
  const initial = loadJSON<FinancialInput & { score?: number; result?: ReturnType<typeof scoreFinancial> }>(STORAGE_KEYS.financial, {} as FinancialInput);
  const [data, setData] = useState<FinancialInput>(initial);
  const [result, setResult] = useState(initial.result ?? null);
  const [aiResult, setAiResult] = useState<any | null>((initial as any).aiResult ?? null);
  const [aiBusy, setAiBusy] = useState(false);
  const set = <K extends keyof FinancialInput>(k: K, v: FinancialInput[K]) => setData((d) => ({ ...d, [k]: v }));
  const num = (v: string) => (v === "" ? undefined : Number(v));
  const { user } = useAuth();
  const showPeriodWarning = needsStrongerStatementPeriod(data.statementPeriodLength, data.targetCountry);

  const runAI = async () => {
    setAiBusy(true);
    const res = await invokeAI<any>("analyze-bank-statement", { ...data });
    setAiBusy(false);
    if (!res.ok) { toast.error(res.message ?? "AI review unavailable."); return; }
    setAiResult(res.result);
    if (user) await supabase.from("bank_statement_scans").insert({ user_id: user.id, input: data as any, result: res.result as any, score: res.result?.score ?? null, risk: res.result?.risk ?? null });
    else toast.info("Sign in to save your AI result.");
    toast.success("AI review complete");
  };

  return (
    <form
      className="space-y-6"
      onSubmit={async (e) => {
        e.preventDefault();
        if (!data.statementPeriodLength) { toast.error("Please select a statement period length."); return; }
        const r = scoreFinancial(data);
        setResult(r);
        saveJSON(STORAGE_KEYS.financial, { ...data, score: r.score, result: r });
        if (user) {
          const { error } = await supabase.from("bank_statement_scans").insert({
            user_id: user.id, input: data as any, result: r as any, score: r.score, risk: r.risk,
          });
          if (error) console.error(error);
        }
        toast.success(`Financial readiness: ${r.score}/100`);
      }}
    >
      <header>
        <h1 className="text-2xl font-semibold">Bank Statement Scan</h1>
        <p className="text-sm text-muted-foreground">AI-assisted financial preparation review. Upload only non-sensitive documents.</p>
      </header>
      <div className="flex items-start gap-2 rounded-md border border-risk/30 bg-risk/5 p-3 text-sm">
        <ShieldAlert className="h-4 w-4 text-risk mt-0.5 shrink-0" />
        <p>Never upload bank passwords, card numbers, CVV, online banking login details, or private access codes.</p>
      </div>

      <div className="rounded-xl border bg-card p-5 grid sm:grid-cols-2 gap-4">
        <Field label="Sponsor name"><Input value={data.sponsorName ?? ""} onChange={(e) => set("sponsorName", e.target.value)} /></Field>
        <Field label="Sponsor relation"><Input value={data.sponsorRelation ?? ""} onChange={(e) => set("sponsorRelation", e.target.value)} /></Field>
        <Field label="Bank name"><Input value={data.bankName ?? ""} onChange={(e) => set("bankName", e.target.value)} /></Field>
        <Field label="Statement period (dates)"><Input placeholder="e.g. Jan–Jun 2026" value={data.statementPeriod ?? ""} onChange={(e) => set("statementPeriod", e.target.value)} /></Field>
        <Field label="Statement period length">
          <Select value={data.statementPeriodLength ?? ""} onValueChange={(v) => set("statementPeriodLength", v)}>
            <SelectTrigger><SelectValue placeholder="Select period length..." /></SelectTrigger>
            <SelectContent>
              {STATEMENT_PERIODS.map((p) => <SelectItem key={p} value={p}>{p}</SelectItem>)}
            </SelectContent>
          </Select>
        </Field>
        <Field label="Target country"><Input value={data.targetCountry ?? ""} onChange={(e) => set("targetCountry", e.target.value)} placeholder="e.g. Italy" /></Field>
        {showPeriodWarning && (
          <div className="sm:col-span-2 flex items-start gap-2 rounded-md border border-warning/40 bg-warning/10 p-3 text-sm">
            <ShieldAlert className="h-4 w-4 text-warning mt-0.5 shrink-0" />
            <p>Most study visa cases need recent, continuous bank history, commonly 3 to 6 months. Some countries may request more.</p>
          </div>
        )}
        <Field label="Currency"><Input value={data.currency ?? ""} onChange={(e) => set("currency", e.target.value)} placeholder="EUR" /></Field>
        <Field label="Opening balance"><Input type="number" value={data.openingBalance ?? ""} onChange={(e) => set("openingBalance", num(e.target.value))} /></Field>
        <Field label="Closing balance"><Input type="number" value={data.closingBalance ?? ""} onChange={(e) => set("closingBalance", num(e.target.value))} /></Field>
        <Field label="Average balance"><Input type="number" value={data.averageBalance ?? ""} onChange={(e) => set("averageBalance", num(e.target.value))} /></Field>

        <Field label="Monthly expenses estimate"><Input type="number" value={data.monthlyExpenses ?? ""} onChange={(e) => set("monthlyExpenses", num(e.target.value))} /></Field>

      </div>

      {result && (
        <div className="rounded-xl border bg-card p-5 grid md:grid-cols-[auto_1fr] gap-5 brovi-fade-up">
          <div className="flex flex-col items-center gap-2">
            <ScoreRing value={result.score} />
            <RiskBadge score={result.score} />
            <Button type="button" size="sm" variant="outline" onClick={runAI} disabled={aiBusy} className="mt-2">
              {aiBusy ? <Loader2 className="h-4 w-4 mr-1 animate-spin" /> : <Sparkles className="h-4 w-4 mr-1" />}
              {aiResult ? "Re-run AI review" : "AI review"}
            </Button>
          </div>
          <div className="space-y-3 text-sm">
            <Row label="Funding gap estimate" value={result.gap ? `${result.gap.toLocaleString()} ${data.currency ?? ""}` : "None"} />
            <Row label="Balance consistency" value={result.consistency} />
            <Row label="Large deposit explanation" value={result.depositQuality} />
            <Row label="Sponsor relationship clarity" value={result.sponsorClarity} />
            <Row label="Source of funds clarity" value={result.sourceClarity} />
            {result.missing.length > 0 && (
              <div>
                <p className="font-medium">Missing financial evidence</p>
                <ul className="mt-1 list-disc list-inside text-muted-foreground">{result.missing.map((m) => <li key={m}>{m}</li>)}</ul>
              </div>
            )}
            <div>
              <p className="font-medium">Safe next steps</p>
              <ol className="mt-1 space-y-1 text-muted-foreground">
                {result.nextSteps.map((s, i) => <li key={s} className="flex gap-2"><span className="h-5 w-5 rounded-full bg-primary/10 text-primary text-xs grid place-items-center">{i + 1}</span>{s}</li>)}
              </ol>
            </div>
            <DisclaimerBanner />
            {aiResult && (
              <div className="rounded-lg border border-primary/30 bg-primary/5 p-4 space-y-2">
                <p className="font-medium flex items-center gap-2"><Sparkles className="h-4 w-4 text-primary" /> AI-assisted review {aiResult.score != null && <span className="text-muted-foreground">· {aiResult.score}/100</span>}</p>
                {aiResult.missing?.length > 0 && <div><p className="font-medium">Missing</p><ul className="list-disc list-inside text-muted-foreground">{aiResult.missing.map((m: string) => <li key={m}>{m}</li>)}</ul></div>}
                {aiResult.nextSteps?.length > 0 && <div><p className="font-medium">Next steps</p><ol className="list-decimal list-inside text-muted-foreground">{aiResult.nextSteps.map((s: string) => <li key={s}>{s}</li>)}</ol></div>}
                {aiResult.warnings?.length > 0 && <div><p className="font-medium text-warning">Warnings</p><ul className="list-disc list-inside text-muted-foreground">{aiResult.warnings.map((s: string) => <li key={s}>{s}</li>)}</ul></div>}
              </div>
            )}
          </div>
        </div>
      )}
    </form>
  );
}

const ATTACH_STATUSES = ["Available", "Missing", "Needs translation", "Needs legalization", "Needs update", "Not applicable"] as const;
type AttachStatus = (typeof ATTACH_STATUSES)[number];

const SUGGESTIONS = [
  "Passport","National ID","Transcript","Degree certificate","School certificate","CV","Motivation letter",
  "Admission letter","Pre-enrollment confirmation","Tuition fee proof","Scholarship letter",
  "Sponsor employment proof","Sponsor business proof","Bank statement","Birth certificate","Family certificate",
  "Police clearance","Accommodation proof","Hotel booking","Flight reservation","Travel insurance",
  "Business invitation","Conference invitation","Medical appointment letter","Visa application form",
  "Appointment confirmation","Translation","Legalization","Other document",
];

type Attachment = {
  id: string;
  documentName: string;
  fileName: string;
  fileSize: number;
  mimeType: string;
  status: AttachStatus;
  uploadedAt: string;
  storagePath?: string;
  badge: "Attached" | "Saved" | "Local only" | "Failed";
};

const MAX_PDF = 20 * 1024 * 1024;
const MAX_IMG = 10 * 1024 * 1024;

function fmtSize(n: number) {
  if (n < 1024) return `${n} B`;
  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`;
  return `${(n / 1024 / 1024).toFixed(2)} MB`;
}

function DocumentsPage() {
  const { user } = useAuth();
  const [documentName, setDocumentName] = useState("");
  const [file, setFile] = useState<File | null>(null);
  const [attachments, setAttachments] = useState<Attachment[]>([]);
  const [busy, setBusy] = useState(false);
  const [destination, setDestination] = useState("");
  const [purpose, setPurpose] = useState("");
  const [aiResult, setAiResult] = useState<any | null>(null);
  const [aiBusy, setAiBusy] = useState(false);
  const [result, setResult] = useState<ReturnType<typeof scoreDocuments> | null>(null);

  // Load existing attachments for signed-in users
  useEffect(() => {
    if (!user) return;
    supabase.from("uploaded_documents").select("*").eq("user_id", user.id).order("created_at", { ascending: false })
      .then(({ data }) => {
        if (!data) return;
        setAttachments(data.map((r: any) => ({
          id: r.id,
          documentName: r.document_category ?? r.file_name,
          fileName: r.file_name,
          fileSize: r.file_size ?? 0,
          mimeType: r.file_type ?? "",
          status: (r.scan_status as AttachStatus) ?? "Available",
          uploadedAt: r.created_at,
          storagePath: r.storage_path ?? undefined,
          badge: r.storage_path ? "Saved" : "Local only",
        })));
      });
  }, [user?.id]);

  const filtered = useMemo(() => {
    if (!documentName) return [] as string[];
    const q = documentName.toLowerCase();
    return SUGGESTIONS.filter((s) => s.toLowerCase().includes(q) && s.toLowerCase() !== q).slice(0, 6);
  }, [documentName]);

  const validate = (f: File): string | null => {
    const isPdf = f.type === "application/pdf" || f.name.toLowerCase().endsWith(".pdf");
    const isImg = f.type.startsWith("image/");
    if (isPdf && f.size > MAX_PDF) return "PDF files must be 20MB or smaller.";
    if (isImg && f.size > MAX_IMG) return "Image files must be 10MB or smaller.";
    if (!isPdf && !isImg && f.size > MAX_PDF) return "File is too large.";
    return null;
  };

  const attach = async () => {
    if (!documentName.trim()) { toast.error("Please type a document name or type."); return; }
    if (!file) { toast.error("Please choose a file to attach."); return; }
    const err = validate(file);
    if (err) { toast.error(err); return; }

    setBusy(true);
    const localId = crypto.randomUUID();
    const now = new Date().toISOString();
    const base: Attachment = {
      id: localId, documentName: documentName.trim(), fileName: file.name,
      fileSize: file.size, mimeType: file.type || "application/octet-stream",
      status: "Available", uploadedAt: now, badge: user ? "Attached" : "Local only",
    };
    setAttachments((p) => [base, ...p]);

    if (user) {
      try {
        const safe = file.name.replace(/[^A-Za-z0-9._-]+/g, "_");
        const path = `${user.id}/general/${Date.now()}-${safe}`;
        const { error: upErr } = await supabase.storage.from("brovi-documents").upload(path, file, {
          contentType: file.type || undefined, upsert: false,
        });
        if (upErr) throw upErr;
        const { data: ins, error: insErr } = await supabase.from("uploaded_documents").insert({
          user_id: user.id,
          file_name: file.name,
          file_size: file.size,
          file_type: file.type,
          document_category: documentName.trim(),
          storage_path: path,
          scan_status: "Available",
        }).select("id").maybeSingle();
        if (insErr) throw insErr;
        setAttachments((p) => p.map((a) => a.id === localId ? { ...a, id: ins?.id ?? a.id, storagePath: path, badge: "Saved" } : a));
      } catch (e) {
        console.error(e);
        setAttachments((p) => p.map((a) => a.id === localId ? { ...a, badge: "Failed" } : a));
        toast.error("Could not save to private storage. The attachment is kept locally.");
      }
    } else {
      toast.info("Sign in to save documents securely.");
    }
    setDocumentName("");
    setFile(null);
    setBusy(false);
  };

  const remove = async (a: Attachment) => {
    setAttachments((p) => p.filter((x) => x.id !== a.id));
    if (user && a.storagePath) {
      try { await supabase.storage.from("brovi-documents").remove([a.storagePath]); } catch { /* ignore */ }
      try { await supabase.from("uploaded_documents").delete().eq("id", a.id); } catch { /* ignore */ }
    }
  };

  const setStatus = async (a: Attachment, s: AttachStatus) => {
    setAttachments((p) => p.map((x) => x.id === a.id ? { ...x, status: s } : x));
    if (user && a.storagePath) {
      await supabase.from("uploaded_documents").update({ scan_status: s }).eq("id", a.id);
    }
  };

  const runScan = async () => {
    const statuses: Record<string, string> = {};
    attachments.forEach((a) => { statuses[a.documentName] = a.status; });
    const r = scoreDocuments(statuses as any);
    setResult(r);
    toast.success(`Document readiness: ${r.score}/100`);
  };

  const runAI = async () => {
    setAiBusy(true);
    const payload = {
      destination,
      travelPurpose: purpose,
      attachments: attachments.map((a) => ({
        documentName: a.documentName, fileName: a.fileName, mimeType: a.mimeType,
        fileSize: a.fileSize, status: a.status, hasStoredFile: !!a.storagePath,
      })),
    };
    const res = await invokeAI<any>("analyze-document-scan", payload);
    setAiBusy(false);
    if (!res.ok) { toast.error(res.message ?? "AI review unavailable."); return; }
    setAiResult(res.result);
    if (user) await supabase.from("document_scans").insert({
      user_id: user.id, input: payload as any, result: res.result as any,
      score: res.result?.score ?? null, risk: res.result?.risk ?? null,
    });
    toast.success("AI review complete");
  };

  const badgeClass = (b: Attachment["badge"]) =>
    b === "Saved" ? "bg-success/10 text-success border-success/30" :
    b === "Attached" ? "bg-primary/10 text-primary border-primary/30" :
    b === "Failed" ? "bg-risk/10 text-risk border-risk/30" :

                </li>
              ))}
            </ul>
          )}
        </section>
      </div>

      {attachments.length > 0 && (
        <section className="rounded-xl border bg-card overflow-x-auto">
          <table className="w-full text-sm min-w-[640px]">
            <thead className="bg-secondary/60 text-muted-foreground">
              <tr>
                <th className="text-left p-3">Document name</th>
                <th className="text-left p-3">File name</th>
                <th className="text-left p-3">Size</th>
                <th className="text-left p-3">Status</th>
                <th className="text-left p-3">Uploaded at</th>
                <th className="text-left p-3">Action</th>
              </tr>
            </thead>

                <tr key={a.id} className="border-t">
                  <td className="p-3 font-medium">{a.documentName}</td>
                  <td className="p-3 text-muted-foreground">{a.fileName}</td>
                  <td className="p-3 text-muted-foreground">{fmtSize(a.fileSize)}</td>
                  <td className="p-3">
                    <Select value={a.status} onValueChange={(v) => setStatus(a, v as AttachStatus)}>
                      <SelectTrigger className="w-[170px]"><SelectValue /></SelectTrigger>
                      <SelectContent>{ATTACH_STATUSES.map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}</SelectContent>
                    </Select>
                  </td>
                  <td className="p-3 text-muted-foreground">{new Date(a.uploadedAt).toLocaleDateString()}</td>
                  <td className="p-3"><button type="button" onClick={() => remove(a)} className="text-risk text-sm hover:underline inline-flex items-center gap-1"><X className="h-3 w-3" /> Remove</button></td>
                </tr>
              ))}
            </tbody>
          </table>
        </section>
      )}

      <div className="flex flex-wrap gap-2">
        <Button type="button" onClick={runScan} disabled={attachments.length === 0}>Run document scan</Button>
        <Button type="button" variant="outline" onClick={runAI} disabled={attachments.length === 0 || aiBusy}>
          {aiBusy ? <Loader2 className="h-4 w-4 mr-1 animate-spin" /> : <Sparkles className="h-4 w-4 mr-1" />}
          AI review
        </Button>
      </div>

      {result && (
        <div className="rounded-xl border bg-card p-5 grid md:grid-cols-[auto_1fr] gap-5 brovi-fade-up">
          <div className="flex flex-col items-center gap-2">
            <ScoreRing value={result.score} />
            <RiskBadge score={result.score} />
          </div>
          <div className="space-y-4">
            <ListBlock title="Missing documents" items={result.missing} tone="risk" />
            <ListBlock title="Urgent issues" items={result.urgent} tone="risk" />
            <ListBlock title="Translation notes" items={result.translation} />
            <ListBlock title="Legalization notes" items={result.legalization} />
            <ListBlock title="Step-by-step correction plan" items={result.actionPlan} numbered />
            {aiResult && (
              <div className="rounded-lg border border-primary/30 bg-primary/5 p-4 space-y-2 text-sm">
                <p className="font-medium flex items-center gap-2"><Sparkles className="h-4 w-4 text-primary" /> AI-assisted review {aiResult.score != null && <span className="text-muted-foreground">· {aiResult.score}/100</span>}</p>
                {(aiResult.missing ?? []).length > 0 && <ListBlock title="AI missing" items={aiResult.missing} tone="risk" />}
                {(aiResult.urgent ?? []).length > 0 && <ListBlock title="AI urgent" items={aiResult.urgent} tone="risk" />}
                {(aiResult.actionPlan ?? []).length > 0 && <ListBlock title="AI action plan" items={aiResult.actionPlan} numbered />}
                {(aiResult.strengths ?? []).length > 0 && <ListBlock title="Strengths" items={aiResult.strengths} />}
                <p className="text-xs text-muted-foreground flex items-start gap-1.5"><CheckCircle2 className="h-3.5 w-3.5 mt-0.5 text-success shrink-0" /> Some file content could not be fully read. The review used your document names, statuses, destination, travel purpose, and available extracted text.</p>
              </div>
            )}
            <DisclaimerBanner />
          </div>
        </div>
      )}
    </div>
  );
}

function ListBlock({ title, items, tone, numbered }: { title: string; items: string[]; tone?: "risk"; numbered?: boolean }) {
  if (!items || !items.length) return null;
  return (
    <div>
      <h3 className="font-medium mb-2">{title}</h3>
      <ol className="space-y-1 text-sm text-muted-foreground">
        {items.map((s, i) => (
          <li key={s + i} className="flex items-start gap-2">
            {numbered ? <span className="h-5 w-5 rounded-full bg-primary/10 text-primary text-xs grid place-items-center shrink-0">{i + 1}</span> : <span className={`h-1.5 w-1.5 rounded-full mt-2 ${tone === "risk" ? "bg-risk" : "bg-primary"}`} />}
            <span>{s}</span>
          </li>
        ))}
      </ol>
    </div>
  );

type Range = "Daily" | "Weekly" | "Monthly" | "Yearly";
function ProgressAnalytics({ scores }: { scores: { program: number; document: number; financial: number; interview: number } }) {
  const [range, setRange] = useState<Range>("Weekly");
  const factor = useMemo(() => ({ Daily: 0.5, Weekly: 1, Monthly: 1.1, Yearly: 1.2 }[range]), [range]);
  const bars = [
    { label: "Program", value: Math.min(100, Math.round(scores.program * factor)) },
    { label: "Documents", value: Math.min(100, Math.round(scores.document * factor)) },
    { label: "Financial", value: Math.min(100, Math.round(scores.financial * factor)) },
    { label: "Interview", value: Math.min(100, Math.round(scores.interview * factor)) },
  ];
  return (
    <div className="rounded-xl border bg-card p-5 lg:col-span-2">
      <div className="flex items-center justify-between gap-3 flex-wrap">
        <div className="flex items-center gap-2">
          <BarChart3 className="h-4 w-4 text-primary" />
          <h2 className="font-semibold">Progress analytics</h2>
        </div>
        <div className="inline-flex rounded-md border p-0.5 bg-background text-xs">
          {(["Daily","Weekly","Monthly","Yearly"] as Range[]).map((r) => (
            <button key={r} onClick={() => setRange(r)}
              className={`px-2.5 py-1 rounded ${range===r ? "bg-secondary text-foreground" : "text-muted-foreground hover:text-foreground"}`}>{r}</button>
          ))}
        </div>
      </div>
      <p className="text-xs text-muted-foreground mt-1">Track your preparation progress across modules.</p>
      <div className="mt-4 space-y-3">
        {bars.map((b) => (
          <div key={b.label}>
            <div className="flex justify-between text-xs mb-1"><span className="text-muted-foreground">{b
